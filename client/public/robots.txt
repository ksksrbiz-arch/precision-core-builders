# Precision Core Builders — crawler policy
# Production domain: https://precisioncorebuilders.com (www 301-redirects to apex)
#
# Intent: the public marketing site (/, /about, /services, /portfolio, /faq,
# /contact, /estimator) is open to search engines AND AI assistants — the
# business explicitly wants AI discoverability. Private app surfaces stay
# blocked for every crawler: /admin, /portal, /auth, /api, /onboarding,
# /dev-login, /callback.

# ==========================================================================
# Default policy — every crawler not matched by a group below.
# Public pages allowed; private app surfaces blocked.
# ==========================================================================
User-agent: *
Allow: /
Disallow: /admin
Disallow: /portal
Disallow: /auth
Disallow: /api
Disallow: /onboarding
Disallow: /dev-login
Disallow: /callback

# ==========================================================================
# AI crawlers — explicitly ALLOWED.
# Precision Core Builders wants to be cited by AI assistants and AI search,
# so reputable AI crawlers get full access to public pages. Private app
# surfaces remain blocked in every group.
# ==========================================================================
